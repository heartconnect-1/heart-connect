function enhanceLocation(){
  if(document.querySelector('.hc10-location'))return;
  const country=field('Country'),locality=field('City / town','City/town','Locality','City','Destination'),region=field('Region','County / state','State / province'),area=field('Area','Neighbourhood','Neighborhood');
  if(!country&&!locality)return;
  const anchor=(country?.closest('label')||locality?.closest('label'));if(!anchor?.parentElement)return;
  const wrap=document.createElement('section');wrap.className='hc10-location';
  wrap.innerHTML='<div class="hc10-location-head"><div><h3>📍 Choose locality</h3><p>Search a city, town, neighbourhood or area. We save the booking location fields, not an exact home GPS position.</p></div></div><div class="hc10-loc-search"><input id="hc10Place" autocomplete="off" placeholder="Search e.g. Kisii, Westlands, Nairobi, Mombasa"><button type="button" id="hc10Search">Search location</button></div><div class="hc10-results" id="hc10Results"></div><div class="hc10-selected" id="hc10Selected" hidden></div><div class="hc10-loc-note">You can still use the normal Country / Region / City / Area fields below if preferred.</div>';
  anchor.parentElement.insertBefore(wrap,anchor);
  const input=wrap.querySelector('#hc10Place'),btn=wrap.querySelector('#hc10Search'),results=wrap.querySelector('#hc10Results'),selected=wrap.querySelector('#hc10Selected');
  async function search(){
    const term=input.value.trim();if(term.length<2){results.innerHTML='<div class="hc10-loc-note">Type at least 2 characters.</div>';return}
    btn.disabled=true;btn.textContent='Searching…';results.innerHTML='';
    try{
      const r=await fetch('/_hc/booking-location-search?q='+encodeURIComponent(term),{headers:{accept:'application/json'}});
      const d=await r.json();if(!r.ok)throw new Error(d.error||'Location search unavailable.');
      results.innerHTML=d.items?.length?d.items.map((x,i)=>'<button class="hc10-result" type="button" data-i="'+i+'"><b>'+esc(x.name||x.locality||x.fullName)+'</b><small>'+esc(x.fullName||[x.area,x.locality,x.region,x.country].filter(Boolean).join(', '))+'</small></button>').join(''):'<div class="hc10-loc-note">No matching places found.</div>';
      results.querySelectorAll('[data-i]').forEach(b=>b.onclick=()=>{const x=d.items[Number(b.dataset.i)];if(!x)return;setValue(country,x.country||country?.value||'');setValue(region,x.region||region?.value||'');setValue(locality,x.locality||x.name||'');setValue(area,x.area||'');selected.hidden=false;selected.textContent='Selected: '+(x.fullName||[x.area,x.locality,x.region,x.country].filter(Boolean).join(', '));results.innerHTML='';update()})
    }catch(e){results.innerHTML='<div class="hc10-loc-note">'+esc(e.message)+'</div>'}finally{btn.disabled=false;btn.textContent='Search location'}
  }
  btn.onclick=search;input.addEventListener('keydown',e=>{if(e.key==='Enter'){e.preventDefault();search()}});
}

function update(){
  const type=bookingType(),isBnb=type==='bnb';
  const ci=field('Check-in'),co=field('Check-out'),rate=field('Preferred nightly rate','Nightly budget / preferred rate'),date=field('Preferred date');
  const country=field('Country'),locality=field('City / town','City/town','Locality','City','Destination');
  const name=field('Full name as shown on ID/passport','Full name','Guest name'),email=field('Email'),phone=field('Phone'),nationality=field('Nationality');
  const doc=field('Identity document','Document type','Upload ID','Verification document');
  const nn=isBnb?nights(ci?.value,co?.value):0,rr=Number(rate?.value||0);
  const stayOk=isBnb?(nn>=1&&nn<=60&&rr>=2500&&rr<=10000):!!date?.value;
  const locOk=!!country?.value&&!!locality?.value;
  const guestOk=!!name?.value&&!!email?.value&&!!phone?.value&&(isBnb?!!nationality?.value:true);
  const verifyOk=isBnb?!!(doc?.value||document.querySelector('input[type=file]')?.files?.length):true;
  const payOk=false;
  const states={stay:stayOk,location:locOk,guest:guestOk,verify:verifyOk,pay:payOk};
  Object.entries(states).forEach(([k,v])=>document.querySelector('[data-hc10-step="'+k+'"]')?.classList.toggle('ready',!!v));
  let panel=document.querySelector('.hc10-ready');
  const grid=document.querySelector('.booking-grid');
  if(!panel&&grid){panel=document.createElement('section');panel.className='hc10-ready';grid.parentElement?.insertBefore(panel,grid.nextSibling)}
  if(panel){
    const checks=[['Dates & '+(isBnb?'nightly budget':'service plan'),stayOk],['Location & locality',locOk],['Guest/contact details',guestOk],[(isBnb?'Identity document':'Request details'),verifyOk]];
    const score=checks.filter(x=>x[1]).length;
    panel.innerHTML='<div class="hc10-ready-head"><h3>Booking readiness</h3><span class="hc10-score">'+score+' / '+checks.length+' ready</span></div><div class="hc10-ready-grid">'+checks.map(x=>'<div class="hc10-check '+(x[1]?'ok':'')+'">'+esc(x[0])+'</div>').join('')+'</div><div class="hc10-review"><div><small>Location</small><b>'+esc([field('Area')?.value,locality?.value,field('Region')?.value,country?.value].filter(Boolean).join(', ')||'Not selected')+'</b></div><div><small>'+(isBnb?'Stay':'Preferred date')+'</small><b>'+(isBnb?(nn?nn+' night'+(nn===1?'':'s'):'Not set'):esc(date?.value||'Not set'))+'</b></div><div><small>'+(isBnb?'Requested total':'Service')+'</small><b>'+(isBnb&&nn&&rr?money(nn*rr):'Calculated after selection')+'</b></div><div><small>Payment</small><b>After provider approval</b></div></div><div class="hc10-safety"><b>Booking safety:</b> room inventory, provider/venue, final price and cancellation terms must be confirmed before the reservation is treated as booked. Payment is not presented as successful until the payment provider confirms it.</div>';
  }
}

function bind(){
  document.querySelectorAll('input,select,textarea').forEach(el=>{if(el.dataset.hc10Bound)return;el.dataset.hc10Bound='1';el.addEventListener('input',update);el.addEventListener('change',update)})
}

addEventListener('DOMContentLoaded',enhance);enhance();
})();
